## Task A2: ArticlesModule

### `backend/src/articles/articles.service.ts`
```ts
import { Injectable, NotFoundException } from '@nestjs/common';
import { PrismaService } from '../prisma/prisma.service';

@Injectable()
export class ArticlesService {
  constructor(private readonly prisma: PrismaService) {}

  async list(params: { category?: string; limit?: number; page?: number }) {
    const page = Math.max(params.page ?? 1, 1);
    const limit = Math.min(params.limit ?? 50, 100);
    const skip = (page - 1) * limit;
    const where = params.category ? { category: params.category as any } : {};
    const [items, total] = await Promise.all([
      this.prisma.article.findMany({ where, skip, take: limit, orderBy: { publishedAt: 'desc' } }),
      this.prisma.article.count({ where }),
    ]);
    return { data: items, meta: { pagination: { page, limit, total, totalPages: Math.ceil(total / limit) } } };
  }

  async bySlug(slug: string) {
    const article = await this.prisma.article.findUnique({ where: { slug } });
    if (!article) throw new NotFoundException('Article not found');
    return article;
  }

  async categories() {
    const counts = await this.prisma.article.groupBy({
      by: ['category'],
      _count: { category: true },
    });
    return counts.map((c) => ({ category: c.category, count: c._count.category }));
  }
}
```

## Task A3: SalariesModule (aggregates from jobs)

### `backend/src/salaries/salaries.service.ts`
```ts
import { Injectable } from '@nestjs/common';
import { PrismaService } from '../prisma/prisma.service';

@Injectable()
export class SalariesService {
  constructor(private readonly prisma: PrismaService) {}

  async snapshot() {
    const [byRole, bySeniority, byRemote, overall] = await Promise.all([
      this.prisma.$queryRawUnsafe<Array<any>>(`
        SELECT t.slug, t.name,
               COUNT(*)::int            as job_count,
               ROUND(AVG(j.salary_min))::int as avg_min,
               ROUND(AVG(j.salary_max))::int as avg_max,
               MIN(j.salary_min)::int   as min_min,
               MAX(j.salary_max)::int   as max_max
        FROM tags t
        JOIN job_tags jt ON jt.tag_id = t.id
        JOIN jobs j ON j.id = jt.job_id
        WHERE j.status = 'live'
          AND j.salary_min IS NOT NULL AND j.salary_max IS NOT NULL
          AND t.category IN ('skill','role')
        GROUP BY t.id
        HAVING COUNT(*) > 0
        ORDER BY avg_max DESC
        LIMIT 25
      `),
      this.prisma.$queryRawUnsafe<Array<any>>(`
        SELECT seniority::text          as seniority,
               COUNT(*)::int            as job_count,
               ROUND(AVG(j.salary_min))::int as avg_min,
               ROUND(AVG(j.salary_max))::int as avg_max
        FROM jobs j
        WHERE j.status = 'live'
          AND j.salary_min IS NOT NULL AND j.salary_max IS NOT NULL
          AND seniority IS NOT NULL
        GROUP BY seniority
        ORDER BY avg_max DESC
      `),
      this.prisma.$queryRawUnsafe<Array<any>>(`
        SELECT (CASE WHEN is_remote THEN 'remote' ELSE 'onsite' END) as kind,
               COUNT(*)::int as job_count,
               ROUND(AVG(salary_min))::int as avg_min,
               ROUND(AVG(salary_max))::int as avg_max
        FROM jobs WHERE status='live' AND salary_min IS NOT NULL AND salary_max IS NOT NULL
        GROUP BY kind
      `),
      this.prisma.$queryRawUnsafe<Array<any>>(`
        SELECT COUNT(*)::int as job_count,
               ROUND(AVG(salary_min))::int as avg_min,
               ROUND(AVG(salary_max))::int as avg_max,
               MIN(salary_min)::int as min_min,
               MAX(salary_max)::int as max_max
        FROM jobs WHERE status='live' AND salary_min IS NOT NULL AND salary_max IS NOT NULL
      `),
    ]);
    return {
      overall: overall[0] ?? null,
      byRole, bySeniority, byRemote,
      updatedAt: new Date().toISOString(),
    };
  }
}

'use client';
import { useEffect, useState } from 'react';

export interface AuthUser {
  id: string;
  email: string;
  role: 'candidate' | 'employer' | 'admin';
  displayName: string | null;
}

interface StoredSession {
  user: AuthUser;
  accessToken: string;
  refreshToken: string;
}

const KEY = 'w3c.session';

export function readSession(): StoredSession | null {
  if (typeof window === 'undefined') return null;
  try {
    const raw = window.localStorage.getItem(KEY);
    return raw ? (JSON.parse(raw) as StoredSession) : null;
  } catch {
    return null;
  }
}

export function writeSession(s: StoredSession) {
  if (typeof window === 'undefined') return;
  window.localStorage.setItem(KEY, JSON.stringify(s));
  window.dispatchEvent(new Event('w3c-session-changed'));
}

export function clearSession() {
  if (typeof window === 'undefined') return;
  window.localStorage.removeItem(KEY);
  window.dispatchEvent(new Event('w3c-session-changed'));
}

export function useSession(): { user: AuthUser | null; loading: boolean } {
  const [session, setSession] = useState<StoredSession | null>(null);
  const [loading, setLoading] = useState(true);
  useEffect(() => {
    setSession(readSession());
    setLoading(false);
    const sync = () => setSession(readSession());
    window.addEventListener('w3c-session-changed', sync);
    window.addEventListener('storage', sync);
    return () => {
      window.removeEventListener('w3c-session-changed', sync);
      window.removeEventListener('storage', sync);
    };
  }, []);
  return { user: session?.user ?? null, loading };
}

export function getAccessToken(): string | null {
  return readSession()?.accessToken ?? null;
}
```

async function get<T>(path: string, init?: RequestInit): Promise<T> {
  const headers: Record<string, string> = { Accept: 'application/json', ...(init?.headers as any ?? {}) };
  if (typeof window !== 'undefined') {
    try {
      const raw = window.localStorage.getItem('w3c.session');
      if (raw) {
        const sess = JSON.parse(raw);
        if (sess?.accessToken) headers.Authorization = `Bearer ${sess.accessToken}`;
      }
    } catch { /* ignore */ }
  }
  const res = await fetch(`${BASE}${path}`, {
    ...init,
    headers,
    next: { revalidate: 30 },
  });
  if (!res.ok) {
    let detail = '';
    try { detail = JSON.stringify(await res.json()); } catch { /* ignore */ }
    throw new Error(`API ${res.status} ${path} ${detail}`);
  }
  return res.json() as Promise<T>;
}
