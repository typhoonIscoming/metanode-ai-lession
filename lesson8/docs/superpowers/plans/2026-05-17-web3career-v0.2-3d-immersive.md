```tsx
'use client';
import { useEffect, useRef } from 'react';

const VERT = /* glsl */ `
attribute vec2 a_pos;
void main() { gl_Position = vec4(a_pos, 0.0, 1.0); }
`;

const FRAG = /* glsl */ `
precision mediump float;
uniform vec2  u_res;
uniform float u_time;

float hash(vec2 p) { return fract(sin(dot(p, vec2(127.1, 311.7))) * 43758.5453); }
float noise(vec2 p) {
  vec2 i = floor(p);
  vec2 f = fract(p);
  vec2 u = f * f * (3.0 - 2.0 * f);
  return mix(
    mix(hash(i + vec2(0.0, 0.0)), hash(i + vec2(1.0, 0.0)), u.x),
    mix(hash(i + vec2(0.0, 1.0)), hash(i + vec2(1.0, 1.0)), u.x),
    u.y);
}
float fbm(vec2 p) {
  float v = 0.0; float a = 0.5;
  for (int i = 0; i < 5; i++) { v += a * noise(p); p *= 2.03; a *= 0.5; }
  return v;
}

void main() {
  vec2 uv = (gl_FragCoord.xy - 0.5 * u_res) / min(u_res.x, u_res.y);
  float t = u_time * 0.06;

  // two slowly-moving noise fields, layered
  float n1 = fbm(uv * 1.6 + vec2(t, -t * 0.7));
  float n2 = fbm(uv * 2.4 + vec2(-t * 0.4, t));
  float n  = mix(n1, n2, 0.5 + 0.5 * sin(t * 0.6));

  // 3-stop crypto palette: violet → cyan → lime
  vec3 violet = vec3(0.659, 0.333, 0.969);   // #a855f7
  vec3 cyan   = vec3(0.024, 0.714, 0.831);   // #06b6d4
  vec3 lime   = vec3(0.518, 0.800, 0.086);   // #84cc16
  vec3 col    = mix(violet, cyan, smoothstep(0.20, 0.55, n));
  col         = mix(col,    lime, smoothstep(0.65, 0.85, n));

  // radial vignette so corners stay dark
  float r = length(uv);
  col *= smoothstep(1.35, 0.35, r);

  // overall opacity so the deep #05070d page bg shows through
  gl_FragColor = vec4(col, 0.32);
}
`;

function compile(gl: WebGLRenderingContext, type: number, src: string): WebGLShader | null {
  const sh = gl.createShader(type);
  if (!sh) return null;
  gl.shaderSource(sh, src);
  gl.compileShader(sh);
  if (!gl.getShaderParameter(sh, gl.COMPILE_STATUS)) {
    console.warn('shader compile failed', gl.getShaderInfoLog(sh));
    gl.deleteShader(sh);
    return null;
  }
  return sh;
}

export function ShaderBg() {
  const ref = useRef<HTMLCanvasElement>(null);

  useEffect(() => {
    if (typeof window === 'undefined') return;
    if (window.matchMedia('(prefers-reduced-motion: reduce)').matches) return;
    const canvas = ref.current;
    if (!canvas) return;
    const gl = canvas.getContext('webgl', { antialias: false, alpha: true, premultipliedAlpha: false });
    if (!gl) return;

    const vs = compile(gl, gl.VERTEX_SHADER, VERT);
    const fs = compile(gl, gl.FRAGMENT_SHADER, FRAG);
    if (!vs || !fs) return;
    const prog = gl.createProgram();
    if (!prog) return;
    gl.attachShader(prog, vs);
    gl.attachShader(prog, fs);
    gl.linkProgram(prog);
    if (!gl.getProgramParameter(prog, gl.LINK_STATUS)) {
      console.warn('program link failed', gl.getProgramInfoLog(prog));
      return;
    }
    gl.useProgram(prog);

    const buf = gl.createBuffer();
    gl.bindBuffer(gl.ARRAY_BUFFER, buf);
    gl.bufferData(gl.ARRAY_BUFFER, new Float32Array([-1, -1, 1, -1, -1, 1, 1, 1]), gl.STATIC_DRAW);
    const aPos = gl.getAttribLocation(prog, 'a_pos');
    gl.enableVertexAttribArray(aPos);
    gl.vertexAttribPointer(aPos, 2, gl.FLOAT, false, 0, 0);

    const uRes  = gl.getUniformLocation(prog, 'u_res');
    const uTime = gl.getUniformLocation(prog, 'u_time');

    function resize() {
      if (!canvas) return;
      const dpr = Math.min(window.devicePixelRatio || 1, 1.5);
      const w = Math.floor(canvas.clientWidth * dpr);
      const h = Math.floor(canvas.clientHeight * dpr);
      if (canvas.width !== w || canvas.height !== h) {
        canvas.width = w;
        canvas.height = h;
        gl!.viewport(0, 0, w, h);
      }
      gl!.uniform2f(uRes, w, h);
    }

    let raf = 0;
    const start = performance.now();
    function frame(t: number) {
      resize();
      gl!.uniform1f(uTime, (t - start) / 1000);
      gl!.drawArrays(gl!.TRIANGLE_STRIP, 0, 4);
      raf = requestAnimationFrame(frame);
    }
    raf = requestAnimationFrame(frame);

    const ro = new ResizeObserver(resize);
    ro.observe(canvas);

    return () => {
      cancelAnimationFrame(raf);
      ro.disconnect();
      gl.deleteProgram(prog);
      gl.deleteShader(vs);
      gl.deleteShader(fs);
      gl.deleteBuffer(buf);
    };
  }, []);

  return (
    <canvas
      ref={ref}
      aria-hidden
      tabIndex={-1}
      className="pointer-events-none fixed inset-0 -z-20 h-full w-full"
    />
  );
}
```

- [ ] **Step 1: Write `frontend/src/components/tilt.tsx`**

```tsx
'use client';
import { useEffect, useRef } from 'react';

interface TiltProps {
  max?: number;
  className?: string;
  children: React.ReactNode;
}

// pure helper, suitable for unit tests
export function computeTilt(
  rect: { left: number; top: number; width: number; height: number },
  clientX: number,
  clientY: number,
  max: number,
): { rotateX: number; rotateY: number; px: number; py: number } {
  const nx = (clientX - rect.left) / rect.width - 0.5;
  const ny = (clientY - rect.top) / rect.height - 0.5;
  return {
    rotateX: -ny * 2 * max,
    rotateY: nx * 2 * max,
    px: ((nx + 0.5) * 100),
    py: ((ny + 0.5) * 100),
  };
}

export function Tilt({ max = 8, className, children }: TiltProps) {
  const ref = useRef<HTMLDivElement>(null);
  const sheenRef = useRef<HTMLDivElement>(null);

  useEffect(() => {
    if (typeof window === 'undefined') return;
    if (window.matchMedia('(prefers-reduced-motion: reduce)').matches) return;
    if (!window.matchMedia('(hover: hover) and (pointer: fine)').matches) return;

    const el = ref.current;
    const sheen = sheenRef.current;
    if (!el) return;

    let raf = 0;
    let next: { x: number; y: number } | null = null;

    function apply() {
      if (!next || !el) { raf = 0; return; }
      const rect = el.getBoundingClientRect();
      const { rotateX, rotateY, px, py } = computeTilt(rect, next.x, next.y, max);
      el.style.transform =
        `perspective(1000px) rotateX(${rotateX.toFixed(2)}deg) rotateY(${rotateY.toFixed(2)}deg)`;
      if (sheen) {
        sheen.style.background =
          `radial-gradient(circle at ${px}% ${py}%, rgba(255,255,255,0.10), transparent 55%)`;
        sheen.style.opacity = '1';
      }
      next = null;
      raf = 0;
    }

    function onMove(e: PointerEvent) {
      next = { x: e.clientX, y: e.clientY };
      if (!raf) raf = requestAnimationFrame(apply);
    }
    function onLeave() {
      if (raf) cancelAnimationFrame(raf);
      raf = 0;
      next = null;
      el!.style.transform = '';
      if (sheen) sheen.style.opacity = '0';
    }

    el.addEventListener('pointermove', onMove);
    el.addEventListener('pointerleave', onLeave);
    return () => {
      el.removeEventListener('pointermove', onMove);
      el.removeEventListener('pointerleave', onLeave);
      if (raf) cancelAnimationFrame(raf);
    };
  }, [max]);

  return (
    <div
      ref={ref}
      className={(className ?? '') + ' relative will-change-transform transition-transform duration-300 ease-out'}
      style={{ transformStyle: 'preserve-3d' }}
    >
      {children}
      <div
        ref={sheenRef}
        aria-hidden
        className="pointer-events-none absolute inset-0 rounded-[inherit] opacity-0 transition-opacity duration-300"
      />
    </div>
  );
}
```

## Task 7: HeroScene component

**Files:**
- Create: `frontend/src/components/hero-scene.tsx`

- [ ] **Step 1: Write `frontend/src/components/hero-scene.tsx`**

```tsx
'use client';
import { Canvas, useFrame } from '@react-three/fiber';
import { Float } from '@react-three/drei';
import { useMemo, useRef } from 'react';
import type { Group, Mesh } from 'three';

function FloatingCube({ position, scale }: { position: [number, number, number]; scale: number }) {
  const meshRef = useRef<Mesh>(null);
  useFrame((_, delta) => {
    if (!meshRef.current) return;
    meshRef.current.rotation.x += delta * 0.6;
    meshRef.current.rotation.y += delta * 0.4;
  });
  return (
    <Float speed={1.6} rotationIntensity={0.2} floatIntensity={0.8}>
      <mesh ref={meshRef} position={position} scale={scale}>
        <boxGeometry args={[0.6, 0.6, 0.6]} />
        <meshBasicMaterial color="#67e8f9" wireframe transparent opacity={0.55} />
      </mesh>
    </Float>
  );
}

function Icosa() {
  const ref = useRef<Mesh>(null);
  useFrame((_, delta) => {
    if (!ref.current) return;
    ref.current.rotation.y += delta * 0.18;
    ref.current.rotation.x += delta * 0.05;
  });
  return (
    <Float speed={1.2} rotationIntensity={0.4} floatIntensity={0.6}>
      <mesh ref={ref}>
        <icosahedronGeometry args={[1.4, 1]} />
        <meshBasicMaterial color="#c084fc" wireframe transparent opacity={0.85} />
      </mesh>
    </Float>
  );
}

function Scene() {
  const cubes = useMemo<Array<{ position: [number, number, number]; scale: number }>>(() => [
    { position: [-2.5,  1.6, -0.5], scale: 0.55 },
    { position: [ 2.4,  1.2,  0.4], scale: 0.40 },
    { position: [-1.8, -1.7,  0.6], scale: 0.50 },
    { position: [ 2.0, -1.3, -0.3], scale: 0.45 },
    { position: [ 0.0,  2.4,  0.8], scale: 0.30 },
    { position: [-2.8, -0.2,  0.5], scale: 0.35 },
    { position: [ 2.8,  0.1, -0.6], scale: 0.42 },
  ], []);

  const groupRef = useRef<Group>(null);
  useFrame((_, delta) => {
    if (!groupRef.current) return;
    groupRef.current.rotation.y += delta * 0.08;
  });

  return (
    <group ref={groupRef}>
      <Icosa />
      {cubes.map((c, i) => (
        <FloatingCube key={i} position={c.position} scale={c.scale} />
      ))}
    </group>
  );
}

export default function HeroScene() {
  if (typeof window !== 'undefined' &&
      window.matchMedia('(prefers-reduced-motion: reduce)').matches) {
    return null;
  }
  return (
    <div aria-hidden className="pointer-events-none absolute inset-y-0 right-0 hidden md:block w-1/2">
      <Canvas
        dpr={[1, 1.5]}
        gl={{ antialias: true, alpha: true }}
        camera={{ position: [0, 0, 5.5], fov: 50 }}
      >
        <Scene />
      </Canvas>
    </div>
  );
}
```

## Task 9: SalaryBars3D component

**Files:**
- Create: `frontend/src/components/salary-bars-3d.tsx`

- [ ] **Step 1: Write `frontend/src/components/salary-bars-3d.tsx`**

```tsx
'use client';
import { Canvas, useFrame } from '@react-three/fiber';
import { Text } from '@react-three/drei';
import { useMemo, useRef } from 'react';
import type { Group } from 'three';

interface Row {
  slug: string;
  name: string;
  avg_max: number;
}

function lerpHex(a: string, b: string, t: number): string {
  const pa = parseInt(a.slice(1), 16);
  const pb = parseInt(b.slice(1), 16);
  const ar = (pa >> 16) & 0xff, ag = (pa >> 8) & 0xff, ab = pa & 0xff;
  const br = (pb >> 16) & 0xff, bg = (pb >> 8) & 0xff, bb = pb & 0xff;
  const r = Math.round(ar + (br - ar) * t);
  const g = Math.round(ag + (bg - ag) * t);
  const bl = Math.round(ab + (bb - ab) * t);
  return '#' + [(r << 16) | (g << 8) | bl].map((v) => v.toString(16).padStart(6, '0'))[0];
}

function Bars({ rows }: { rows: Row[] }) {
  const groupRef = useRef<Group>(null);
  useFrame((_, delta) => {
    if (!groupRef.current) return;
    groupRef.current.rotation.y += delta * 0.08;
  });

  const maxVal = useMemo(() => rows.reduce((m, r) => Math.max(m, r.avg_max), 1), [rows]);
  const count = rows.length;
  const spacing = 0.95;
  const offset  = -((count - 1) * spacing) / 2;

  return (
    <group ref={groupRef}>
      {rows.map((r, i) => {
        const t = r.avg_max / maxVal;
        const height = Math.max(0.15, t * 4);
        const color = lerpHex('#22d3ee', '#c084fc', t);
        return (
          <group key={r.slug} position={[offset + i * spacing, 0, 0]}>
            <mesh position={[0, height / 2, 0]}>
              <boxGeometry args={[0.55, height, 0.55]} />
              <meshBasicMaterial color={color} transparent opacity={0.88} />
            </mesh>
            <mesh position={[0, height / 2, 0]}>
              <boxGeometry args={[0.56, height, 0.56]} />
              <meshBasicMaterial color={color} wireframe transparent opacity={0.55} />
            </mesh>
            <Text
              position={[0, height + 0.35, 0]}
              fontSize={0.22}
              color="#e5e7eb"
              anchorX="center"
              anchorY="middle"
            >
              {`$${Math.round(r.avg_max / 1000)}k`}
            </Text>
            <Text
              position={[0, -0.35, 0]}
              fontSize={0.18}
              color="#9ca3af"
              anchorX="center"
              anchorY="middle"
              rotation={[-Math.PI / 4, 0, 0]}
            >
              {r.name}
            </Text>
          </group>
        );
      })}
      <gridHelper args={[count * spacing + 2, count + 2, '#1f2937', '#0f1729']} position={[0, 0, 0]} />
    </group>
  );
}

export default function SalaryBars3D({ rows }: { rows: Row[] }) {
  if (typeof window !== 'undefined' &&
      window.matchMedia('(prefers-reduced-motion: reduce)').matches) {
    return null;
  }
  if (rows.length === 0) return null;
  const top10 = rows.slice(0, 10);
  return (
    <div aria-hidden className="glass rounded-xl overflow-hidden h-[360px]">
      <Canvas
        dpr={[1, 1.5]}
        gl={{ antialias: true, alpha: true }}
        camera={{ position: [0, 3.6, 9], fov: 45 }}
      >
        <Bars rows={top10} />
      </Canvas>
    </div>
  );
}
```
